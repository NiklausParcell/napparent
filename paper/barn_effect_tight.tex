\begin{abstract}
Given a tabular sample with scalar outcome $Y^{(n)}\in\mathbb{R}$ (label, probability, or model output), we discretize each feature to bins, then accumulate co-occurrence counts $C_{uv}$ and outcome sums $W_{uv}$ over all unordered column pairs $(i,j)$ and encoded bin pairs $(u,v)$.
The artifact is a sparse weighted graph $\bm{\Phi}$ with $\Phi_{uv}=(W_{uv}/C_{uv})\log_{10}(C_{uv})$ when $C_{uv}>1$, else $0$.
Training and inference are chunk-additive; storage is $O(p^2B^2)$ in bin depth $B$.
We record a.s.\ convergence of a scaled $\Phi_{uv}$ to the stratum conditional mean under i.i.d.\ sampling and \emph{fixed} quantization, and note identifiability limits for empty or singleton cells \cite{Agresti2013,Manski2003}.
No experiments; full discussion appears in \texttt{barn\_effect.tex}.
\end{abstract}

\section{Problem and scope}
We target \emph{descriptive} row-level signals from empirical co-occurrence structure, complementary to model-based explainers (e.g.\ \cite{Lundberg2017,Ribeiro2016}) and to ICE-style curve views \cite{Goldstein2015}.
Streaming sufficiency follows \cite{Gama2014,Bifet2018}.
This note states definitions and two propositions only.

\section{Preliminaries}
Dataset $\mathcal{D}=\{(\bm{x}^{(n)},Y^{(n)})\}_{n=1}^N$, features $i\in[p]$, values $X_i^{(n)}$.
Column pairs $\mathcal{C}=\{(i,j):1\le i<j\le p\}$.
Optional chunks $\mathcal{D}_k$ partition $\mathcal{D}$.

\section{Phase 1: quantization}
\begin{definition}[Bin schema]
\label{def:bin-tight}
Fix $B_i$.
\emph{Numerical $i$:} range $r_i=\max_n X_i^{(n)}-\min_n X_i^{(n)}$; round to $s_i$ decimal places via $s_i=\max(0,4-\lfloor\log_{10} r_i\rfloor)$; take top-$B_i$ rounded frequencies as interior split candidates, replace largest candidate by $\max_n X_i^{(n)}$, prepend $\min_n X_i^{(n)}$, form $B_i$ interval bins.
\emph{Categorical $i$:} keep top-$B_i$ levels; map others to $\varepsilon$.
Write $\mathcal{B}=\{(\mathcal{L}_i,\ell^{(i)})\}$, $\ell^{(i)}:\mathrm{dom}(X_i)\to\mathcal{L}_i$.
Disjoint column encodings identify $\mathcal{V}=\bigcup_i\mathcal{L}_i$ with $[V]$ so cross-column bin pairs are unambiguous.
\end{definition}

\section{Phase 2: accumulation and graph}
For encoded labels $u,v \in [V]$, define $\kappa(u,v):=(\min\{u,v\},\max\{u,v\})$.
Fix $\alpha \in [0,1)$ (default $0.01$); for chunk $\mathcal{D}_k$ with $n_k=|\mathcal{D}_k|$, set $\theta_k:=\lfloor \alpha n_k\rfloor$.
For $(i,j)\in\mathcal{C}$ and each chunk, let $c_\kappa^{(k)}$ be the chunk-local count of rows with $\kappa(\ell^{(i)}(X_i^{(n)}),\ell^{(j)}(X_j^{(n)}))=\kappa$.
If $c_\kappa^{(k)}>\theta_k$, add to sparse $(W,C)$:
\begin{equation}
\label{eq:WC-tight}
\Delta W_{\kappa}^{(k)}=\!\!\sum_{\substack{n\in\mathcal{D}_k:\\ \kappa(\ell^{(i)}(X_i^{(n)}),\ell^{(j)}(X_j^{(n)}))=\kappa}}\!\! Y^{(n)},\qquad
\Delta C_{\kappa}^{(k)}=c_\kappa^{(k)};
\end{equation}
otherwise $\Delta W_{\kappa}^{(k)}=\Delta C_{\kappa}^{(k)}=0$.
Summing over chunks gives global $W_{\kappa},C_{\kappa}$ keyed by $\kappa(u,v)$; when $\theta_k=0$ for all $k$, totals are partition-invariant.
\begin{definition}[Interaction graph]
$\Phi_{\kappa}=0$ if $C_{\kappa}\in\{0,1\}$, else $\Phi_{\kappa}=(W_{\kappa}/C_{\kappa})\log_{10}(C_{\kappa})$.
\end{definition}

\section{Inference}
For row $n$, $\psi_{ij}^{(n)}=\Phi_{\kappa(\ell^{(i)}(X_i^{(n)}),\ell^{(j)}(X_j^{(n)}))}$.
Let $m_c:=p-1$.
\[
A_c^{(n)}=\frac{1}{m_c}\sum_{j\neq c}\psi_{cj}^{(n)},
\qquad
\delta_c^{(n)}=A_c^{(n)}-\bar{Y},\quad \bar{Y}=\frac{1}{N}\sum_n Y^{(n)}.
\]
Global score $\bar{\delta}_c=\frac{1}{N}\sum_n\delta_c^{(n)}$.
Attributions do \emph{not} satisfy an efficiency axiom $\sum_c\delta_c^{(n)}=Y^{(n)}-\bar{Y}$ \cite{Lundberg2017}.

\section{Properties}
\begin{remark}
When $\theta_k=0$ for all chunks, \eqref{eq:WC-tight} is scheduled additively and totals match one pass over $\mathcal{D}$; otherwise borderline pairs may depend on partition.
\end{remark}

\begin{proposition}[Complexity]
Training: $O(p^2N)$ time, $O(p^2B^2)$ sparse keys.
Inference per row: $O(p^2)$ lookups.
\end{proposition}

\begin{remark}[Fixed $\mathcal{B}$ for limits]
\label{rem:fixed-tight}
Proposition~\ref{prop:conv-tight} treats $\mathcal{B},\ell^{(i)}$ as nonrandom; Definition~\ref{def:bin-tight} instantiates empirical $\mathcal{B}_N$ on the sample.
Joint limits for $(\mathcal{B}_N,\bm{\Phi})$ are not pursued.
\end{remark}

\begin{proposition}[Scaled a.s.\ limit for $\Phi_{uv}$]
\label{prop:conv-tight}
If $(\bm{x}^{(n)},Y^{(n)})$ are i.i.d., $\mathbb{E}|Y|<\infty$, $\mathcal{B}$ fixed, and $\pi=P(\ell^{(i)}(X_i)=u,\ell^{(j)}(X_j)=v)>0$, then
\[
\frac{\Phi_{uv}}{\log_{10}(N\pi)}\xrightarrow{\mathrm{a.s.}} \mathbb{E}\bigl[Y\mid \ell^{(i)}(X_i)=u,\ell^{(j)}(X_j)=v\bigr].
\]
\end{proposition}
\begin{proof}[Sketch]
$C_{uv}/N\to\pi$ a.s.; $\log_{10}C_{uv}-\log_{10}N\to\log_{10}\pi$ a.s.; $W_{uv}/C_{uv}\to\mathbb{E}[Y\mid u,v]$ on the stratum.
Combine and rescale.
\end{proof}

\begin{remark}[Support]
Zero $\Phi_{\kappa}$ does not imply $\mathbb{E}[Y\mid u,v]=0$: $C_{\kappa}=0$ means unseen; $C_{\kappa}=1$ is zeroed by construction.
Sparse cells --- partial identification \cite{Manski2003}; inference cautions \cite{Agresti2013}.
Column attribution averages over all $m_c=p-1$ partners; absent pairs contribute zero.
\end{remark}

\section{Implementation (minimal)}
Phase~1 builds $\mathcal{B}$; Phase~2 accumulates $(W,C)$ under $\kappa(u,v)$ with threshold $\theta_k$, then forms $\bm{\Phi}$; Phase~3 computes $\psi$, $A_c$, and $\delta_c$.
$Y^{(n)}$ may be raw $y$, class probability, or $\hat f(\bm{x}^{(n)})$.

\section{Related work (one pass)}
Co-occurrence of discretized items parallels frequent-pattern mining \cite{Agrawal1994} but $W$ supplies a supervised conditional mean, not binary support alone.
The $\log_{10} C$ factor resembles frequency dampers in IR \cite{SparckJones1972}.

\section*{Conclusion}
The Barn Effect is a discrete pairwise sufficient-statistic summary $\bm{\Phi}$ plus row scores $\bm{\Delta}$, streamable and $O(p^2B^2)$-bounded.
The main theoretical loose end is extending Proposition~\ref{prop:conv-tight} to empirical $\mathcal{B}_N$ (Remark~\ref{rem:fixed-tight}).
